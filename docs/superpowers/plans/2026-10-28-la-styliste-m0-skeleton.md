# la-styliste M0 Skeleton Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An Expo SDK 58 app that opens in Expo Go on the iPhone with four empty Turkish tabs, a dark theme, a working settings store, and the guards (network allowlist, public-repo commit gate) every later milestone builds on.

**Architecture:** Expo Router app in `app/` of the repo. SQLite through a four-method `Db` interface (expo-sqlite on the phone, `node:sqlite` in tests) with append-only migrations. A small theme system: one light and one dark theme object, `useTheme()` and `makeStyles()`, never static token imports in components.

**Tech Stack:** Expo SDK 58 (React Native 0.88, React 19.3), expo-router (`expo-router/js-tabs`), expo-sqlite, TypeScript ~6.0, Vitest 5 on Node 24.

**Spec:** [`docs/superpowers/specs/2026-10-28-la-styliste-design.md`](../specs/2026-10-28-la-styliste-design.md) (§3 M0 row, §4 architecture, §5 settings, §10 migrations, §12 gates).

`app/src/data/db.ts`:

```ts
export type SqlParam = string | number | null;

/** The only surface repositories use. One adapter per runtime: expo-sqlite on the phone, node:sqlite in tests. */
export interface Db {
  exec(sql: string): Promise<void>;
  run(sql: string, params?: SqlParam[]): Promise<{ lastInsertRowId: number; changes: number }>;
  all<T>(sql: string, params?: SqlParam[]): Promise<T[]>;
  first<T>(sql: string, params?: SqlParam[]): Promise<T | null>;
}

/** Runs `work` between BEGIN and COMMIT; any error rolls the whole step back and is rethrown. */
export async function inTransaction<T>(db: Db, work: () => Promise<T>): Promise<T> {
  await db.exec('BEGIN');
  try {
    const result = await work();
    await db.exec('COMMIT');
    return result;
  } catch (e) {
    await db.exec('ROLLBACK');
    throw e;
  }
}
```

`app/src/data/expoDb.ts`:

```ts
import * as SQLite from 'expo-sqlite';
import type { Db, SqlParam } from './db';

/** The database file name. Changing it would orphan every install's data. */
const DB_NAME = 'la-styliste.db';

export async function openAppDb(): Promise<Db> {
  const d = await SQLite.openDatabaseAsync(DB_NAME);
  await d.execAsync('PRAGMA journal_mode = WAL');
  return {
    exec: (sql: string) => d.execAsync(sql),
    run: async (sql: string, params: SqlParam[] = []) => {
      const r = await d.runAsync(sql, params);
      return { lastInsertRowId: r.lastInsertRowId, changes: r.changes };
    },
    all: <T,>(sql: string, params: SqlParam[] = []) => d.getAllAsync<T>(sql, params),
    first: <T,>(sql: string, params: SqlParam[] = []) => d.getFirstAsync<T>(sql, params),
  };
}
```

`app/src/data/errors.ts`:

```ts
/** The database was written by a newer build of the app. Opening it with older migrations could lose data, so the app refuses. */
export class SchemaTooNewError extends Error {
  readonly found: number;
  readonly known: number;

  constructor(found: number, known: number) {
    super(`Database schema v${found} is newer than this app (v${known})`);
    this.name = 'SchemaTooNewError';
    this.found = found;
    this.known = known;
  }
}

/** A settings patch held a value its parser rejects. Nothing of that patch was written. */
export class InvalidSettingError extends Error {
  readonly setting: string;

  constructor(setting: string) {
    super(`Invalid value for setting "${setting}"`);
    this.name = 'InvalidSettingError';
    this.setting = setting;
  }
}
```

```ts
import { DatabaseSync } from 'node:sqlite';
import type { Db, SqlParam } from '../db';

function wrap(d: DatabaseSync): Db {
  return {
    async exec(sql: string) {
      d.exec(sql);
    },
    async run(sql: string, params: SqlParam[] = []) {
      const r = d.prepare(sql).run(...params);
      return { lastInsertRowId: Number(r.lastInsertRowid), changes: Number(r.changes) };
    },
    async all<T>(sql: string, params: SqlParam[] = []) {
      return d.prepare(sql).all(...params) as unknown as T[];
    },
    async first<T>(sql: string, params: SqlParam[] = []) {
      return (d.prepare(sql).get(...params) ?? null) as unknown as T | null;
    },
  };
}

/** In-memory SQLite on Node's built-in driver. Tests only. */
export function openMemoryDb(): Db {
  return wrap(new DatabaseSync(':memory:'));
}

/** A database exactly as an install left it at schema `version`: the first `version` migrations, applied. Tests only. */
export async function dbAtVersion(version: number, migrations: readonly string[]): Promise<Db> {
  const db = openMemoryDb();
  for (const sql of migrations.slice(0, version)) await db.exec(sql);
  await db.exec(`PRAGMA user_version = ${version}`);
  return db;
}
```

`app/src/data/migrations.ts`:

```ts
import { inTransaction, type Db } from './db';
import { SchemaTooNewError } from './errors';

/** Append only: index i upgrades schema version i to i + 1. Never edit an entry that has reached a phone. */
export const MIGRATIONS: readonly string[] = [
  // M0: key-value settings. Values are JSON text; the key list lives in `settings.ts`.
  `CREATE TABLE settings (
     key TEXT PRIMARY KEY,
     value TEXT NOT NULL
   );`,
];

/**
 * Brings `db` to the last version of `migrations`, one transaction per step, so a failing step leaves the
 * database at the version before it. Refuses a database that is newer than `migrations`. Returns the version reached.
 */
export async function migrate(db: Db, migrations: readonly string[] = MIGRATIONS): Promise<number> {
  const row = await db.first<{ user_version: number }>('PRAGMA user_version');
  let version = row?.user_version ?? 0;
  if (version > migrations.length) throw new SchemaTooNewError(version, migrations.length);
  while (version < migrations.length) {
    const next = version + 1;
    const sql = migrations[version];
    await inTransaction(db, async () => {
      await db.exec(sql);
      await db.exec(`PRAGMA user_version = ${next}`);
    });
    version = next;
  }
  return version;
}
```

```ts
import { inTransaction, type Db } from './db';
import { InvalidSettingError } from './errors';

/** Which look to draw: the phone's own, or always light, or always dark. */
export type ThemeMode = 'system' | 'light' | 'dark';

export interface AppSettings {
  /** Light, dark or the phone's choice. */
  themeMode: ThemeMode;
}

/** A fresh install's settings, and what a missing or corrupt stored value falls back to. Dark by default (spec §11). */
export const DEFAULT_SETTINGS: AppSettings = {
  themeMode: 'dark',
};

/**
 * The database key of each setting. This list is also the backup allowlist (spec §5, §10): a name that is
 * not here is never read, written or exported. Later milestones add their settings here.
 */
export const SETTING_KEYS: { readonly [K in keyof AppSettings]: string } = {
  themeMode: 'theme_mode',
};

/** One parser per setting: stored JSON is untrusted, so each returns a valid value or null. */
const PARSERS: { readonly [K in keyof AppSettings]: (raw: unknown) => AppSettings[K] | null } = {
  themeMode: (raw) => (raw === 'system' || raw === 'light' || raw === 'dark' ? raw : null),
};

const NAMES = Object.keys(SETTING_KEYS) as (keyof AppSettings)[];

function parseStored<K extends keyof AppSettings>(name: K, text: string): AppSettings[K] | null {
  try {
    return PARSERS[name](JSON.parse(text));
  } catch {
    return null;
  }
}

export async function loadSettings(db: Db): Promise<AppSettings> {
  const rows = await db.all<{ key: string; value: string }>('SELECT key, value FROM settings');
  const stored = new Map(rows.map((r) => [r.key, r.value]));
  const settings: AppSettings = { ...DEFAULT_SETTINGS };
  for (const name of NAMES) {
    const text = stored.get(SETTING_KEYS[name]);
    const value = text === undefined ? null : parseStored(name, text);
    if (value !== null) (settings as Record<keyof AppSettings, unknown>)[name] = value;
  }
  return settings;
}

/** Writes every setting of `patch` in one transaction. Names that are not settings are skipped; an invalid value writes nothing. */
export async function saveSettings(db: Db, patch: Partial<AppSettings>): Promise<void> {
  const writes: [string, string][] = [];
  for (const name of NAMES) {
    if (!Object.hasOwn(patch, name)) continue;
    const value = patch[name];
    if (PARSERS[name](value) === null) throw new InvalidSettingError(name);
    writes.push([SETTING_KEYS[name], JSON.stringify(value)]);
  }
  if (writes.length === 0) return;
  await inTransaction(db, async () => {
    for (const [key, value] of writes) {
      await db.run('INSERT INTO settings (key, value) VALUES (?, ?) ON CONFLICT(key) DO UPDATE SET value = excluded.value', [key, value]);
    }
  });
}
```

```ts
import { describe, expect, it } from 'vitest';
import { ALL_THEMES } from './registry';

/** WCAG 2.x relative luminance of a `#RRGGBB` colour. */
function luminance(hex: string): number {
  const channel = (i: number) => {
    const c = parseInt(hex.slice(i, i + 2), 16) / 255;
    return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  };
  return 0.2126 * channel(1) + 0.7152 * channel(3) + 0.0722 * channel(5);
}

/** WCAG contrast ratio between two `#RRGGBB` colours. */
function contrast(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
}

describe.each(ALL_THEMES.map((t) => [t.id, t] as const))('theme %s', (_id, { colors }) => {
  it('uses #RRGGBB everywhere', () => {
    for (const value of Object.values(colors)) expect(value).toMatch(/^#[0-9A-F]{6}$/i);
  });

  it('keeps body and secondary text at 4.5:1 on every surface', () => {
    for (const surface of [colors.bg, colors.surface, colors.surfaceRaised]) {
      expect(contrast(colors.text, surface)).toBeGreaterThanOrEqual(4.5);
      expect(contrast(colors.textDim, surface)).toBeGreaterThanOrEqual(4.5);
    }
  });

  it('keeps faint text at 3:1 on the background, and accent and error text readable', () => {
    expect(contrast(colors.textFaint, colors.bg)).toBeGreaterThanOrEqual(3);
    expect(contrast(colors.onAccent, colors.accent)).toBeGreaterThanOrEqual(4.5);
    expect(contrast(colors.dangerText, colors.bg)).toBeGreaterThanOrEqual(4.5);
  });
});
```

```ts
import type { ThemeMode } from '../../data/settings';
import type { Theme } from './types';

/** What `useColorScheme()` reports: the phone's scheme, or `unspecified` or nothing when it has none. */
export type SystemScheme = 'light' | 'dark' | 'unspecified' | null | undefined;

/** Light or dark: an explicit mode wins, `system` follows the phone, and a phone with no preference means dark. */
export function resolveAppearance(mode: ThemeMode, system: SystemScheme): Theme['appearance'] {
  if (mode === 'light' || mode === 'dark') return mode;
  return system === 'light' ? 'light' : 'dark';
}

/**
 * What to hand `Appearance.setColorScheme`, which pins native alerts, sheets and pickers too. In `system`
 * mode the pin is lifted (`unspecified`), or the app would keep echoing the last explicit choice.
 */
export function colorSchemeFor(mode: ThemeMode): 'light' | 'dark' | 'unspecified' {
  return mode === 'system' ? 'unspecified' : mode;
}
```

`app/src/ui/themes/perTheme.ts`:

```ts
import type { Theme } from './types';

/** Runs `factory` once per theme object and returns the same result on every later call with it. */
export function perTheme<T>(factory: (theme: Theme) => T): (theme: Theme) => T {
  const results = new WeakMap<Theme, { value: T }>();
  return (theme) => {
    let hit = results.get(theme);
    if (!hit) {
      hit = { value: factory(theme) };
      results.set(theme, hit);
    }
    return hit.value;
  };
}
```

```tsx
import { createContext, useContext, useEffect, type ReactNode } from 'react';
import { Appearance, useColorScheme } from 'react-native';
import type { ThemeMode } from '../../data/settings';
import { colorSchemeFor, resolveAppearance } from './appearance';
import { dark } from './dark';
import { perTheme } from './perTheme';
import { getTheme } from './registry';
import type { Theme } from './types';

export type { Theme } from './types';

const ThemeContext = createContext<Theme>(dark);

/** Draws `children` in the theme `mode` asks for; `system` follows the phone and changes live when it switches. */
export function ThemeProvider({ mode, children }: { mode: ThemeMode; children: ReactNode }) {
  const system = useColorScheme();
  const theme = getTheme(resolveAppearance(mode, system));
  useEffect(() => {
    Appearance.setColorScheme(colorSchemeFor(mode));
  }, [mode]);
  return <ThemeContext.Provider value={theme}>{children}</ThemeContext.Provider>;
}

/** The active theme. Every colour, size and spacing value a component draws with comes from here. */
export function useTheme(): Theme {
  return useContext(ThemeContext);
}

/**
 * `const useStyles = makeStyles((t) => StyleSheet.create({ ... }))`, then `const styles = useStyles()`.
 * The factory runs once per theme, however many components use it.
 */
export function makeStyles<T>(factory: (theme: Theme) => T): () => T {
  const stylesFor = perTheme(factory);
  return function useStyles() {
    return stylesFor(useTheme());
  };
}
```

```tsx
import { createContext, useCallback, useContext, useEffect, useMemo, useState, type ReactNode } from 'react';
import type { Db } from '../data/db';
import { openAppDb } from '../data/expoDb';
import { migrate } from '../data/migrations';
import { DEFAULT_SETTINGS, loadSettings, saveSettings, type AppSettings } from '../data/settings';

export interface AppData {
  /** The database is open, migrated and read. */
  ready: boolean;
  /** Why opening, migrating or reading failed; the root layout shows the recovery screen while it is set. */
  error: string | null;
  settings: AppSettings;
  updateSettings(patch: Partial<AppSettings>): Promise<void>;
  /** Opens, migrates and reads the database again. */
  retry(): void;
}

const AppDataContext = createContext<AppData | null>(null);

export function AppDataProvider({ children }: { children: ReactNode }) {
  const [db, setDb] = useState<Db | null>(null);
  const [settings, setSettings] = useState<AppSettings>(DEFAULT_SETTINGS);
  const [error, setError] = useState<string | null>(null);
  const [attempt, setAttempt] = useState(0);

  useEffect(() => {
    let live = true;
    setError(null);
    (async () => {
      const opened = await openAppDb();
      await migrate(opened);
      const loaded = await loadSettings(opened);
      if (!live) return;
      setSettings(loaded);
      setDb(opened);
    })().catch((e: unknown) => {
      if (live) setError(e instanceof Error ? e.message : String(e));
    });
    return () => {
      live = false;
    };
  }, [attempt]);

  const updateSettings = useCallback(
    async (patch: Partial<AppSettings>) => {
      if (db === null) throw new Error('The database is not open yet');
      await saveSettings(db, patch);
      setSettings((current) => ({ ...current, ...patch }));
    },
    [db],
  );

  const retry = useCallback(() => setAttempt((n) => n + 1), []);

  const value = useMemo<AppData>(
    () => ({ ready: db !== null, error, settings, updateSettings, retry }),
    [db, error, settings, updateSettings, retry],
  );
  return <AppDataContext.Provider value={value}>{children}</AppDataContext.Provider>;
}

export function useAppData(): AppData {
  const value = useContext(AppDataContext);
  if (value === null) throw new Error('useAppData needs AppDataProvider above it');
  return value;
}
```

```ts
import { readdirSync, readFileSync } from 'node:fs';
import { join, relative, sep } from 'node:path';
import { fileURLToPath } from 'node:url';
import { describe, expect, it } from 'vitest';

// Privacy you can check (spec §7, §12): only files under src/net/ may reach a network, and only these hosts.
// This scans the app's own source; it cannot see inside node_modules.

/** The only hosts the app talks to: OpenRouter for the LLM, Open-Meteo for weather and city search. */
const ALLOWED_HOSTS = ['openrouter.ai', 'api.open-meteo.com', 'geocoding-api.open-meteo.com'];

/** Ways JavaScript reaches a network. Any match outside src/net/ fails the build. */
const NETWORK = [
  { what: 'fetch(', re: /(?<![\w$])fetch\s*(?:\?\.\s*)?\(|(?<![\w$])fetch\s*\.\s*(?:call|apply|bind)\b/ },
  { what: 'XMLHttpRequest', re: /\bXMLHttpRequest\b/ },
  { what: 'WebSocket', re: /\bWebSocket\b/ },
  { what: 'EventSource', re: /\bEventSource\b/ },
  { what: 'sendBeacon', re: /\bsendBeacon\b/ },
  { what: 'an HTTP client import', re: /(?:\bfrom\s*|\brequire\s*\(\s*|\bimport\s*\(?\s*)['"](?:axios|ky|expo-network)(?:\/[^'"]*)?['"]/ },
  // Every expo-file-system download or upload API has one of these words in its name. Fails closed: reword a comment that uses them.
  { what: 'a download or upload', re: /\b\w*(?:download|upload)\w*\b/i },
  { what: 'an http(s) URL', re: /['"`]https?:\/\//i },
];

function networkUses(source: string): string[] {
  return NETWORK.filter(({ re }) => re.test(source)).map(({ what }) => what);
}

const SRC = fileURLToPath(new URL('./', import.meta.url));
const NET = join(SRC, 'net') + sep;

function sourceFiles(dir: string): string[] {
  return readdirSync(dir, { withFileTypes: true }).flatMap((e) => {
    const path = join(dir, e.name);
    if (e.isDirectory()) return sourceFiles(path);
    return /\.tsx?$/.test(e.name) && !/\.test\.tsx?$/.test(e.name) ? [path] : [];
  });
}

const rel = (f: string) => relative(SRC, f).split(sep).join('/');

describe('the network detector', () => {
  it.each([
    ['fetch', `await fetch('x')`, ['fetch(']],
    ['fetch.call', `fetch.call(globalThis, url)`, ['fetch(']],
    ['XMLHttpRequest', `new XMLHttpRequest()`, ['XMLHttpRequest']],
    ['WebSocket', `new WebSocket(url)`, ['WebSocket']],
    ['axios', `import axios from 'axios';`, ['an HTTP client import']],
    ['File.downloadFileAsync', `await File.downloadFileAsync(url, dir)`, ['a download or upload']],
    ['createUploadTask', `file.createUploadTask(url)`, ['a download or upload']],
    ['an image from the web', `<Image source={{ uri: 'https://example.com/a.png' }} />`, ['an http(s) URL']],
  ])('catches %s', (_name, source, found) => {
    expect(networkUses(source)).toEqual(found);
  });

  it('lets through words that only look like network code and the local file and share APIs', () => {
    expect(networkUses(`const refetch = () => {}; refetch();`)).toEqual([]);
    expect(networkUses(`import { File, Paths } from 'expo-file-system'; new File(Paths.document, 'x');`)).toEqual([]);
    expect(networkUses(`import * as Sharing from 'expo-sharing'; await Sharing.shareAsync(uri);`)).toEqual([]);
  });
});

describe('network code in the app', () => {
  const files = sourceFiles(SRC);

  it('scans a real tree', () => {
    // A wrong root would pass by scanning nothing.
    expect(files.length).toBeGreaterThan(15);
  });

  it('appears only under src/net/', () => {
    const found = files
      .filter((f) => !f.startsWith(NET))
      .flatMap((f) => networkUses(readFileSync(f, 'utf8')).map((u) => `${rel(f)}: ${u}`));
    expect(found).toEqual([]);
  });

  it('talks only to the allowed hosts inside src/net/', () => {
    const hosts = files
      .filter((f) => f.startsWith(NET))
      .flatMap((f) => [...readFileSync(f, 'utf8').matchAll(/https?:\/\/([a-z0-9.-]+)/gi)].map((m) => m[1].toLowerCase()));
    expect(hosts.filter((h) => !ALLOWED_HOSTS.includes(h))).toEqual([]);
  });
});
```

`app/src/tooling/publicCheck.ts` (plain TypeScript with no enums or parameter properties, so Node can strip its types when the script imports it):

```ts
/** One staged file as the commit gate sees it. */
export interface StagedFile {
  path: string;
  bytes: number;
  binary: boolean;
}

/** Binary files above this size are refused outside app/assets/: photos and backups do not belong in a public repo. */
export const MAX_BINARY_BYTES = 200 * 1024;

/** Text that must never be committed: an OpenRouter key, a private key block. */
const SECRETS: readonly { what: string; re: RegExp }[] = [
  { what: 'an OpenRouter key', re: /sk-or-[A-Za-z0-9-]{8,}/ },
  { what: 'a private key', re: /-----BEGIN [A-Z ]*PRIVATE KEY-----/ },
];

/** Images and archives, allowed only under app/assets/. */
const MEDIA = /\.(jpe?g|png|webp|heic|zip)$/i;

/** Where the app's own icon and splash files live; the only place media may be committed. */
const ASSETS_DIR = 'app/assets/';

/** What is wrong with a staged change; empty when it may be committed to the public repo. */
export function findProblems(files: readonly StagedFile[], addedLines: readonly string[]): string[] {
  const problems: string[] = [];
  for (const f of files) {
    if (f.path.startsWith(ASSETS_DIR)) continue;
    if (MEDIA.test(f.path)) problems.push(`${f.path}: image or archive outside ${ASSETS_DIR}`);
    else if (f.binary && f.bytes > MAX_BINARY_BYTES) problems.push(`${f.path}: binary over 200 KB outside ${ASSETS_DIR}`);
  }
  for (const line of addedLines) {
    for (const { what, re } of SECRETS) if (re.test(line)) problems.push(`an added line contains ${what}`);
  }
  return problems;
}
```

`app/scripts/check-public.mjs`:

```js
// Commit gate for the public repo (spec §12): refuses staged photos, archives, large binaries and secrets.
// Node strips the types of the imported .ts file (Node 22.18+ / 24).
import { execFileSync } from 'node:child_process';
import { findProblems } from '../src/tooling/publicCheck.ts';

/** Generous output limit: a large staged diff must not crash the gate. */
const MAX_OUTPUT = 256 * 1024 * 1024;

const root = execFileSync('git', ['rev-parse', '--show-toplevel'], { encoding: 'utf8' }).trim();
const git = (...args) => execFileSync('git', ['-C', root, ...args], { encoding: 'utf8', maxBuffer: MAX_OUTPUT });

const paths = git('diff', '--cached', '--name-only', '--diff-filter=ACMR', '-z').split('\0').filter(Boolean);
const files = paths.map((path) => ({
  path,
  bytes: Number(git('cat-file', '-s', `:${path}`).trim()),
  // numstat prints "-<TAB>-" for a binary file.
  binary: git('diff', '--cached', '--numstat', '--', path).startsWith('-\t-\t'),
}));
const added = git('diff', '--cached', '-U0', '--no-color')
  .split('\n')
  .filter((line) => line.startsWith('+') && !line.startsWith('+++'));

const problems = findProblems(files, added);
if (problems.length > 0) {
  console.error(`check:public refused the commit:\n${problems.map((p) => `  - ${p}`).join('\n')}`);
  process.exit(1);
}
console.log(`check:public: ${files.length} staged file(s) clean`);
```
